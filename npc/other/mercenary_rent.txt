-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[Comerciante de Artigos para Mercenários]";
	mes "Olá, vendo artigos que os";
	mes "Mercenários podem usar.";
	mes "Está procurando algo";
	mes "em específico?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[Comerciante de Artigos para Mercenários]";
	mes getitemname(.@item[.@m]);
	mes "custa " + .@str_cost$ + " zeny cada.";
	mes "Quantos você quer?";
	next;
	input .@input;
	if (!.@input) {
		mes "[Comerciante de Artigos para Mercenários]";
		mes "Mudou de ideia?";
		mes "Tudo bem, volte sempre que";
		mes "quiser comprar Poções para Mercenários.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[Comerciante de Artigos para Mercenários]";
		mes "Desculpe, mas você";
		mes "só pode comprar até";
		mes "10.000 dessas poções";
		mes "de uma vez. Por favor, insira";
		mes "um número de 1 a 10.000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[Comerciante de Artigos para Mercenários]";
		mes "Desculpe, mas você não tem";
		mes "zeny suficiente para comprar";
		mes "tantas poções. Estarei";
		mes "aqui quando estiver pronto";
		mes "para comprar algo";
		mes "para seus Mercenários.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[Comerciante de Artigos para Mercenários]";
		mes "Se eu te der tantas";
		mes "poções, você não conseguirá";
		mes "carregá-las. Por favor, volte";
		mes "após liberar espaço";
		mes "no seu Inventário.";
		close;
	}
	mes "[Comerciante de Artigos para Mercenários]";
	if (.@input == 1)
		mes "Aqui está seu " + getitemname(.@item[.@m]) + ".";
	else {
		mes "Aqui está, exatamente";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "Obrigado, e volte sempre que";
	mes "precisar de mais poções";
	mes "para seus Mercenários.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Verificador]";
	mes "Por favor, insira uma senha.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Verificador]";
	if (.@i == -2)
		mes "Erro.";
	else if (.@i == 0)
		mes "Número incorreto";
	else if (.@i == 1) {
		mes "Selecione uma opção.";
		next;
		switch( select( "Desativar NPC de Mercenários","Ativar NPC de Mercenários" ) ) {
		case 1:
			mes "NPCs desativados.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "NPCs ativados.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
