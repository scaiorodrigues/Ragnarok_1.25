function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "Você quer adicionar um Slot em um "+ getitemname(.@item_id) +"?";
	mes "Por favor, traga-me ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000 ") : "") + "e meus ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +".000 zeny^000000 de taxa.";
	mes "Ah, e não esqueça de trazer a "+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Posso tentar adicionar um slot agora se você tiver os itens e zeny necessários.";
		mes "Porém, saiba que há uma chance de eu falhar.";
		mes "Por isso, preciso lhe avisar com antecedência...";
		next;
		mes "[Seiyablem]";
		mes "Se falhar, você perde apenas os ^FF0000materiais e o zeny^000000 — a Arma e suas Cartas ficam intactas.";
	}
	else // armor
	{
		mes "Se você tiver todos os materiais, minha taxa em zeny e a Armadura, podemos prosseguir com a tentativa de Adição de Slot.";
		mes "Mas antes, devo avisá-lo sobre o risco.";
		next;
		mes "[Seiyablem]";
		mes "Se falhar, você perde apenas os ^FF0000materiais e o zeny^000000 — a Armadura, o refino e as Cartas ficam intactos.";
	}
	mes "Você ainda quer tentar adicionar um Slot?";
	next;
	switch(select("Tentar Adicionar Slot:Cancelar"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "Muito bem, que o trabalho comece!";
				mes "É melhor rezar por um resultado bem-sucedido.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Ótimo, parece ter dado certo.";
					mes "Ficou muito bem feito. Parabéns!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "Até a próxima, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Ah! ...Sinto muito, falhou.";
					mes "No entanto, não foi culpa minha.";
					mes "É uma questão de sorte, determinada pelos deuses.";
					mes "Não fique tão desanimado, tente novamente.";
					// [Hardcore] A falha destruia o equipamento, o refino e as
					// cartas. Agora so os materiais e o zeny sao consumidos.
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Seu equipamento esta intacto — perdemos apenas os materiais.";
					mes "Boa sorte na próxima vez!";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "Gostaria de prosseguir com a tentativa, mas você está com falta de alguns itens.";
				mes "Tem certeza de que possui o equipamento, os materiais necessários e o zeny?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Precisa de tempo para pensar, hein?";
			mes "Certo, posso entender.";
			mes "Só lembre-se: a vida não tem graça se você sempre joga pelo seguro~";
			close;
	}
}
